; - Fixed: streaming updated roughly once every two seconds instead of following
;   the feed. Excel applies Application.RTD.ThrottleInterval to every RTD server
;   and ships it at 2000ms, so a cell repainted at most twice a minute per
;   symbol however fast the add-in pushed. Broker feeds run at roughly 1 to 11
;   updates per second, so most updates were discarded and a slow moving symbol

;   oa_ws_connect(). Version 1.0.4 was not affected because it bypassed the RTD

; - Renamed: oa_quote is now oa_field. oa_quote was a prefix of oa_quotes and
;   differed only by a trailing s, so typing =oa_quotes( opened Excel's formula
;   autocomplete with oa_quote highlighted and Tab or Enter silently completed

[Code]
const
  SettingsKey = 'Software\OpenAlgo\ExcelAddIn';

var
  BitnessResolved: Boolean;
  BitnessIs64: Boolean;

{ Detects the Office major version from the per-user hive. }
function GetExcelVersion(var Version: string): Boolean;
begin
  Result := True;
  if RegKeyExists(HKEY_CURRENT_USER, 'Software\Microsoft\Office\16.0\Excel') then
    Version := '16.0'
  else if RegKeyExists(HKEY_CURRENT_USER, 'Software\Microsoft\Office\15.0\Excel') then
    Version := '15.0'
  else
  begin
    Version := '';
    Result := False;
  end;
end;

{
  Determines whether the installed Excel is 64-bit.

  The 1.0.4 implementation shelled out to PowerShell and tested
  Is64BitOperatingSystem together with an Excel.exe file size above 2 MB.
  Neither half is a bitness test: the first describes Windows rather than
  Office, and Excel.exe is roughly 50 to 80 MB at both bitnesses, so the size
  comparison is always true. The net effect was that 32-bit Excel on 64-bit
  Windows received the 64-bit XLL and failed to load.

  Click-to-Run records the real platform, and MSI installs can be told apart by
  which registry view holds Excel's InstallRoot: only 64-bit Office writes to
  the native view. The answer is cached because Inno calls Check functions once
  per file entry.
}
function IsExcel64Bit(): Boolean;
var
  Platform: String;
begin
  if BitnessResolved then
  begin
    Result := BitnessIs64;
    Exit;
  end;

  BitnessIs64 := False;

  if not IsWin64 then
    // 32-bit Windows cannot run 64-bit Office.
    BitnessIs64 := False
  else if RegQueryStringValue(HKLM64, 'SOFTWARE\Microsoft\Office\ClickToRun\Configuration', 'Platform', Platform) then
    BitnessIs64 := (CompareText(Platform, 'x64') = 0)
  else if RegKeyExists(HKLM64, 'SOFTWARE\Microsoft\Office\16.0\Excel\InstallRoot') then
    BitnessIs64 := True
  else if RegKeyExists(HKLM64, 'SOFTWARE\Microsoft\Office\15.0\Excel\InstallRoot') then
    BitnessIs64 := True
  else
    // Nothing in the native view means a 32-bit Office under WOW6432Node, or no
    // Office at all. Installing the 32-bit add-in is the safe default.
    BitnessIs64 := False;

  BitnessResolved := True;
  Result := BitnessIs64;
end;

{
  Registers the XLL in the first free OPEN slot.

  Excel enumerates add-ins as OPEN, OPEN1, OPEN2 and so on. Writing a fixed
  OPEN overwrites whichever add-in already holds slot 0. This walks the list,
  reuses our own entry when reinstalling, and records the slot so uninstall
  removes exactly that value and nothing else.
}
procedure RegisterAddIn();
var
  Version, RegPath, XllPath, ValueName, Existing: String;
  Index: Integer;
begin
  if not GetExcelVersion(Version) then
    Version := '16.0';

  RegPath := 'Software\Microsoft\Office\' + Version + '\Excel\Options';

  if IsExcel64Bit then
    XllPath := ExpandConstant('{userappdata}\Microsoft\AddIns\OpenAlgo-AddIn64-packed.xll')
  else
    XllPath := ExpandConstant('{userappdata}\Microsoft\AddIns\OpenAlgo-AddIn-packed.xll');

  Index := 0;
  ValueName := 'OPEN';
  while RegQueryStringValue(HKEY_CURRENT_USER, RegPath, ValueName, Existing) do
  begin
    // Reuse our own slot on reinstall rather than accumulating entries.
    if Pos('openalgo-addin', Lowercase(Existing)) > 0 then
      Break;
    Index := Index + 1;
    ValueName := 'OPEN' + IntToStr(Index);
  end;

  RegWriteStringValue(HKEY_CURRENT_USER, RegPath, ValueName, '/R "' + XllPath + '"');

  RegWriteStringValue(HKEY_CURRENT_USER, SettingsKey, 'OpenValueName', ValueName);
  RegWriteStringValue(HKEY_CURRENT_USER, SettingsKey, 'OfficeVersion', Version);
end;

procedure CurStepChanged(CurStep: TSetupStep);
begin
  if CurStep = ssPostInstall then
    RegisterAddIn;
end;

procedure CurUninstallStepChanged(CurUninstallStep: TUninstallStep);
var
  Version, RegPath, ValueName, ConfigDir: String;
begin
  if CurUninstallStep <> usPostUninstall then
    Exit;

  if not RegQueryStringValue(HKEY_CURRENT_USER, SettingsKey, 'OfficeVersion', Version) then
    if not GetExcelVersion(Version) then
      Version := '16.0';

  if not RegQueryStringValue(HKEY_CURRENT_USER, SettingsKey, 'OpenValueName', ValueName) then
    ValueName := 'OPEN';

  RegPath := 'Software\Microsoft\Office\' + Version + '\Excel\Options';

  // Delete only our value. The surrounding key holds the user's Excel settings
  // and any other add-in registrations.
  RegDeleteValue(HKEY_CURRENT_USER, RegPath, ValueName);
  RegDeleteKeyIncludingSubkeys(HKEY_CURRENT_USER, SettingsKey);

  // config.json holds the OpenAlgo API key in clear text and websocket.log can
  // hold session detail, so offer to remove them rather than leaving them behind.
  ConfigDir := ExpandConstant('{localappdata}\OpenAlgo');
  if DirExists(ConfigDir) then
  begin
    if MsgBox('Remove saved OpenAlgo settings and logs?' + #13#10 + #13#10 +
              'This deletes your stored API key from:' + #13#10 + ConfigDir,
              mbConfirmation, MB_YESNO) = IDYES then
      DelTree(ConfigDir, True, True, True);
  end;
end;
